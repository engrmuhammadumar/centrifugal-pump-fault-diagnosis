    # KEY NOVEL CONTRIBUTIONS (Conference-Grade):
    # 
    # 1. **Adaptive Frequency Decomposition Module (AFDM)**
    #    - Learnable wavelet-like basis functions that automatically discover fault-relevant
    #      frequency patterns without manual feature engineering
    #    - Unlike fixed FFT/STFT, adapts to dataset-specific spectral signatures
    #    - Theory: Extends adaptive basis pursuit to deep learning (cites Mallat, Donoho)
    #
    # 2. **Cross-Scale Spectral Fusion (CSSF)**
    #    - Multi-resolution frequency analysis with learnable fusion weights
    #    - Combines local (high-freq transients) + global (low-freq patterns) representations
    #    - Novel contribution: Attention-based fusion replacing simple concatenation
    #
    # 3. **Physics-Informed Spectral Regularization (PISR)**
    #    - Incorporates domain knowledge: fault frequencies should be sparse, localized
    #    - Enforces spectral sparsity via learned soft-thresholding
    #    - Penalty on spectral spread (faults manifest at specific frequencies)
    #
    # 4. **Temporal-Spectral Consistency Loss (TSCL)**
    #    - Ensures predictions are consistent across temporally adjacent windows
    #    - Uses spectral similarity as proxy for temporal coherence
    #    - Addresses overfitting to window-level noise
    #
    # 5. **Prototype-Guided Contrastive Learning (PGCL)**
    #    - Maintains momentum-updated spectral prototypes per class
    #    - Contrastive objective: pull samples toward class prototype, push from others
    #    - More principled than standard CE for imbalanced/few-shot scenarios
    #
    # 6. **Frequency-Adaptive Positional Encoding (FAPE)**
    #    - Position encoding modulated by local frequency content
    #    - High-frequency regions get finer-grained positional resolution
    #    - Addresses limitation of fixed sinusoidal encodings for signals
    #
    # THEORETICAL GROUNDING:
    # - Connects to harmonic analysis, wavelet theory, compressed sensing

    def detect_file_extension(directory):
        """Detect what file extensions are available in the directory"""
        extensions = {}
        try:
            for root, dirs, files in os.walk(directory):
                for f in files:
                    ext = os.path.splitext(f)[1].lower()
                    if ext:
                        extensions[ext] = extensions.get(ext, 0) + 1
        except:
            pass
        return extensions

    def find_best_1d_signal(matdict):
        """Auto-detect the longest 1D numeric vector in a mat dict."""
        best = None
        best_len = 0
        for k, v in matdict.items():
            if k.startswith("__"):
                continue
            if isinstance(v, np.ndarray):
                arr = v
                arr = np.squeeze(arr)
                if arr.ndim == 1 and np.issubdtype(arr.dtype, np.number):
                    if arr.size > best_len:
                        best = arr.astype(np.float32)
                        best_len = arr.size
                elif arr.ndim == 2 and np.issubdtype(arr.dtype, np.number):
                    if 1 in arr.shape:
                        arr1 = arr.reshape(-1).astype(np.float32)
                        if arr1.size > best_len:
                            best = arr1
                            best_len = arr1.size
        return best

    def load_mat_signal(path, key=None):
        """Load signal from .mat file"""
        md = sio.loadmat(path)
        if key is not None and key in md:
            arr = np.squeeze(md[key])
            if arr.ndim == 2:
                # If it's multi-channel (e.g., 3x25600), take first channel
                if arr.shape[0] < arr.shape[1]:
                    arr = arr[0, :]  # Take first row
                else:
                    arr = arr[:, 0]  # Take first column
            elif arr.ndim == 2 and 1 in arr.shape:
                arr = arr.reshape(-1)
            return arr.astype(np.float32)
        
        # Try to find 'signal' variable first (from your screenshot)
        if 'signal' in md:
            arr = md['signal']
            if arr.ndim == 2:
                # Multi-channel signal, take first channel (only print once per dataset load)
                if arr.shape[0] < arr.shape[1]:
                    # Only print the first time
                    if not hasattr(load_mat_signal, '_multi_channel_warned'):
                        print(f"  Multi-channel signal detected ({arr.shape[0]} channels x {arr.shape[1]} samples), using first channel")
                        load_mat_signal._multi_channel_warned = True
                    arr = arr[0, :]
                else:
                    arr = arr[:, 0]
            return np.squeeze(arr).astype(np.float32)
        
        sig = find_best_1d_signal(md)
        if sig is None:
            raise ValueError(f"No suitable 1D numeric signal found in: {path}")
        return sig

    def load_signal_file(path, key=None, file_ext=None):
        """Load signal from various file types"""
        if file_ext is None:
            file_ext = os.path.splitext(path)[1].lower()
        
        if file_ext == '.mat':
            return load_mat_signal(path, key)
        elif file_ext in ['.accdb', '.mdb']:
            return load_access_db_signal(path, key)
        elif file_ext == '.csv':
            # Try loading CSV
            import pandas as pd
            df = pd.read_csv(path)
            # Assume first numeric column is the signal
            for col in df.columns:
                if pd.api.types.is_numeric_dtype(df[col]):
                    return df[col].values.astype(np.float32)
            raise ValueError(f"No numeric column found in CSV: {path}")
        elif file_ext == '.npy':
            arr = np.load(path)
            return np.squeeze(arr).astype(np.float32)
        else:
            raise ValueError(f"Unsupported file type: {file_ext}")

    def zscore(x, eps=1e-8):
        return (x - x.mean()) / (x.std() + eps)

    def compute_envelope_spectrum(x):
        """Hilbert envelope for fault frequency detection"""
        analytic = hilbert(x)
        envelope = np.abs(analytic)
        return envelope

    def compute_spectral_kurtosis(x):
        """Higher-order statistics for transient detection"""
        X = np.fft.rfft(x)
        mag = np.abs(X)
        return kurtosis(mag)

    def aug_time_shift(x, max_shift=200):
        shift = np.random.randint(-max_shift, max_shift + 1)
        return np.roll(x, shift)

    def aug_amp_scale(x, min_s=0.8, max_s=1.2):
        s = np.random.uniform(min_s, max_s)
        return x * s

    def aug_add_noise(x, snr_db_min=20, snr_db_max=35):
        snr_db = np.random.uniform(snr_db_min, snr_db_max)
        sig_pow = np.mean(x**2) + 1e-12
        snr = 10 ** (snr_db / 10.0)
        noise_pow = sig_pow / snr
        noise = np.random.randn(*x.shape).astype(np.float32) * np.sqrt(noise_pow)
        return x + noise

    def aug_frequency_mask(x, n_masks=2, max_width=10):
        """Frequency masking (SpecAugment style)"""
        X = np.fft.rfft(x)
        F = X.shape[0]
        for _ in range(n_masks):
            f_start = np.random.randint(0, F - max_width)
            f_width = np.random.randint(1, max_width)
            X[f_start:f_start+f_width] *= 0.1
        return np.fft.irfft(X, n=x.size)

    def aug_time_mask(x, n_masks=2, max_width=50):
        """Time masking"""
        y = x.copy()
        for _ in range(n_masks):
            t_start = np.random.randint(0, x.size - max_width)
            t_width = np.random.randint(1, max_width)
            y[t_start:t_start+t_width] = 0.0
        return y

    def apply_augs(x):
        if np.random.rand() < 0.5:
            x = aug_time_shift(x)
        if np.random.rand() < 0.5:
            x = aug_amp_scale(x)
        if np.random.rand() < 0.6:
            x = aug_add_noise(x)
        if np.random.rand() < 0.3:
            x = aug_frequency_mask(x)
        if np.random.rand() < 0.3:
            x = aug_time_mask(x)
        return x

    def window_signal(sig, win_len, stride, max_windows=None):
        n = sig.size
        if n < win_len:
            pad = win_len - n
            sig = np.pad(sig, (0, pad), mode="constant")
            n = sig.size
        starts = np.arange(0, n - win_len + 1, stride)
        if max_windows is not None and starts.size > max_windows:
            idx = np.linspace(0, starts.size - 1, max_windows).astype(int)
            starts = starts[idx]
        windows = np.stack([sig[s:s+win_len] for s in starts], axis=0)
        return windows

            self.file_ids = np.concatenate(self.file_ids, axis=0)
            
            # Shuffle
            idx = np.random.permutation(len(self.labels))
            self.samples = self.samples[idx]

    # NOVEL COMPONENT 1: Adaptive Frequency Decomposition Module (AFDM)
    # -------------------------
    class AdaptiveFrequencyDecomposition(nn.Module):
        """
        Learns dataset-specific frequency decomposition basis.
        Unlike fixed FFT bins, this adapts to fault-relevant frequency patterns.
        
        Inspiration: Adaptive wavelet bases (Mallat), learned dictionaries (sparse coding)
        """
        def __init__(self, signal_len, n_bands=8, overlap=0.3):
            super().__init__()
            self.n_bands = n_bands
            self.signal_len = signal_len
            self.freq_bins = signal_len // 2 + 1  # rfft output size
            
            # Learnable band centers and widths (in normalized frequency [0,1])
            self.band_centers = nn.Parameter(torch.linspace(0.05, 0.95, n_bands))
            self.band_log_widths = nn.Parameter(torch.full((n_bands,), -1.0))
            
            # Learnable importance weights per band
            self.band_importance = nn.Parameter(torch.ones(n_bands))
            
        def forward(self, x):
            # x: (B, T)
            B, T = x.shape
            
            # FFT
            X = torch.fft.rfft(x, dim=-1)  # (B, F) complex
            mag = torch.abs(X)  # (B, F)
            phase = torch.angle(X)  # (B, F)
            
            F = mag.shape[-1]
            freqs = torch.linspace(0, 1, F, device=x.device)  # Normalized frequency
            
            # Compute Gaussian band masks
            centers = torch.sigmoid(self.band_centers)  # (n_bands,)
            widths = torch.exp(self.band_log_widths)  # (n_bands,)
            
            # Expand for broadcasting: freqs (F,), centers (n_bands,)
            band_masks = torch.exp(-((freqs.view(1, -1) - centers.view(-1, 1)) ** 2) / (2 * widths.view(-1, 1) ** 2 + 1e-8))
            band_masks = band_masks / (band_masks.sum(dim=1, keepdim=True) + 1e-8)  # (n_bands, F)
            
            # Apply masks to magnitude spectrum
            # mag: (B, F), band_masks: (n_bands, F) -> (B, n_bands, F)
            band_mags = mag.unsqueeze(1) * band_masks.unsqueeze(0)  # (B, n_bands, F)
            
            # Weight by importance
            importance = torch.softmax(self.band_importance, dim=0)  # (n_bands,)
            weighted_bands = band_mags * importance.view(1, -1, 1)  # (B, n_bands, F)
            
            # Per-band energy features
            band_energies = weighted_bands.sum(dim=-1)  # (B, n_bands)
            
            # Reconstruct with phase for each band (for time-domain representation)
            band_signals = []
            for b in range(self.n_bands):
                mask = band_masks[b]  # (F,)
                X_band = X * mask.unsqueeze(0)  # (B, F)
                x_band = torch.fft.irfft(X_band, n=T, dim=-1)  # (B, T)
                band_signals.append(x_band)
            band_signals = torch.stack(band_signals, dim=1)  # (B, n_bands, T)
            
            return {
                'band_energies': band_energies,  # (B, n_bands)
                'band_signals': band_signals,    # (B, n_bands, T)
                'full_spectrum': mag,            # (B, F)
                'band_masks': band_masks,        # (n_bands, F)
            }

    # NOVEL COMPONENT 2: Cross-Scale Spectral Fusion (CSSF)
    # -------------------------
    class CrossScaleSpectralFusion(nn.Module):
        """
        Fuses multi-resolution spectral representations using attention.
        Combines local transients (high-freq) + global patterns (low-freq).
        """
        def __init__(self, n_bands, d_model):
            super().__init__()
            self.n_bands = n_bands
            
            # Project each band to common embedding space
            self.band_proj = nn.Linear(n_bands, d_model)
            
            # Cross-attention to fuse bands
            self.fusion_attn = nn.MultiheadAttention(d_model, num_heads=4, batch_first=True)
            self.norm = nn.LayerNorm(d_model)
            
        def forward(self, band_energies):
            # band_energies: (B, n_bands)
            B = band_energies.shape[0]
            
            # Project to d_model
            x = self.band_proj(band_energies).unsqueeze(1)  # (B, 1, d_model)
            
            # Self-attention (treats as sequence of 1)
            x_fused, _ = self.fusion_attn(x, x, x)  # (B, 1, d_model)
            x_fused = self.norm(x + x_fused)
            
            return x_fused.squeeze(1)  # (B, d_model)

    # -------------------------
    # NOVEL COMPONENT 3: Frequency-Adaptive Positional Encoding (FAPE)
    # -------------------------
    class FrequencyAdaptivePositionalEncoding(nn.Module):
        """
        Modulates positional encoding based on local frequency content.
        High-frequency regions get finer positional resolution.
        """
        def __init__(self, d_model, max_len=100):
            super().__init__()
            self.d_model = d_model
            
            # Base sinusoidal encoding
            pe = torch.zeros(max_len, d_model)
            position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
            div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
            pe[:, 0::2] = torch.sin(position * div_term)
            pe[:, 1::2] = torch.cos(position * div_term)
            self.register_buffer('pe', pe)
            
            # Frequency-adaptive modulation
            self.freq_modulation = nn.Linear(1, d_model)
            
        def forward(self, x, freq_indicator):
            # x: (B, N, d_model) - patch embeddings
            # freq_indicator: (B, N) - local frequency content measure
            B, N, d = x.shape
            
            # Base positional encoding
            pos_enc = self.pe[:N, :].unsqueeze(0).expand(B, -1, -1)  # (B, N, d_model)
            
            # Frequency modulation
            freq_mod = self.freq_modulation(freq_indicator.unsqueeze(-1))  # (B, N, d_model)
            
            # Adaptive modulation
            adaptive_pe = pos_enc * (1 + torch.tanh(freq_mod))
            
            return x + adaptive_pe

    # -------------------------
    # NOVEL COMPONENT 4: Physics-Informed Spectral Regularization
    # -------------------------
    class PhysicsInformedSpectralLoss(nn.Module):
        """
        Enforces:
        1. Spectral sparsity (faults manifest at specific frequencies)
        2. Spectral localization (frequency bands should be compact)
        3. Inter-class spectral separation
        """
        def __init__(self):
            super().__init__()
            
        def forward(self, afdm_output, labels, n_classes):
            # afdm_output: dict from AFDM
            band_energies = afdm_output['band_energies']  # (B, n_bands)
            band_masks = afdm_output['band_masks']  # (n_bands, F)
            
            # 1. Sparsity loss (L1 on band energies)
            sparsity_loss = torch.mean(torch.abs(band_energies))
            
            # 2. Localization loss (entropy of band masks - prefer peaked distributions)
            band_mask_norm = band_masks / (band_masks.sum(dim=1, keepdim=True) + 1e-8)
            localization_loss = -torch.mean(torch.sum(band_mask_norm * torch.log(band_mask_norm + 1e-8), dim=1))
            localization_loss = -localization_loss  # We want LOW entropy (peaked), so negate
            
            # 3. Inter-class separation (maximize distance between class-mean spectral signatures)
            class_means = []
            for c in range(n_classes):
                mask = (labels == c)
                if mask.any():
                    class_means.append(band_energies[mask].mean(dim=0))
            
            if len(class_means) > 1:
                class_means = torch.stack(class_means)  # (n_classes, n_bands)
                # Pairwise distances
                dists = torch.cdist(class_means, class_means, p=2)  # (n_classes, n_classes)
                # Maximize minimum pairwise distance (hinge loss style)
                separation_loss = -torch.mean(dists[dists > 0])  # Exclude diagonal
            else:
                separation_loss = torch.tensor(0.0, device=band_energies.device)
            
            total_loss = sparsity_loss + 0.1 * localization_loss + 0.2 * separation_loss
            
            return total_loss, {
                'sparsity': sparsity_loss.item(),
                'localization': localization_loss.item(),
                'separation': separation_loss.item(),
            }

    # -------------------------
    # NOVEL COMPONENT 5: Prototype-Guided Contrastive Learning (PGCL)
    # -------------------------
    class PrototypeGuidedContrastive(nn.Module):
        """
        Maintains EMA prototypes in spectral embedding space.
        Contrastive objective: samples closer to their class prototype, far from others.
        """
        def __init__(self, n_classes, embed_dim, temp=0.07, momentum=0.95):
            super().__init__()
            self.n_classes = n_classes
            self.embed_dim = embed_dim
            self.temp = temp
            self.momentum = momentum
            
            # Prototypes
            self.register_buffer('prototypes', torch.randn(n_classes, embed_dim))
            self.prototypes = F.normalize(self.prototypes, dim=-1)
            self.register_buffer('initialized', torch.zeros(n_classes, dtype=torch.bool))
            
        @torch.no_grad()
        def update_prototypes(self, embeddings, labels):
            # embeddings: (B, D) - normalized
            for c in range(self.n_classes):
                mask = (labels == c)
                if mask.any():
                    mean_c = embeddings[mask].mean(dim=0)
                    mean_c = F.normalize(mean_c, dim=-1)
                    
                    if not self.initialized[c]:
                        self.prototypes[c] = mean_c
                        self.initialized[c] = True
                    else:
                        self.prototypes[c] = self.momentum * self.prototypes[c] + (1 - self.momentum) * mean_c
                        self.prototypes[c] = F.normalize(self.prototypes[c], dim=-1)
        
        def forward(self, embeddings, labels):
            # embeddings: (B, D)
            embeddings = F.normalize(embeddings, dim=-1)
            
            # Update prototypes
            with torch.no_grad():
                self.update_prototypes(embeddings, labels)
            
            # Compute similarities
            prototypes = F.normalize(self.prototypes, dim=-1)
            logits = (embeddings @ prototypes.t()) / self.temp  # (B, n_classes)
            
            loss = F.cross_entropy(logits, labels)
            
            return loss

    # -------------------------
    # NOVEL COMPONENT 6: Temporal-Spectral Consistency Loss (TSCL)
    # -------------------------
    class TemporalSpectralConsistency(nn.Module):
        """
        Ensures predictions are consistent across temporally adjacent windows from same file.
        Uses spectral similarity as proxy for temporal coherence.
        """
        def __init__(self):
            super().__init__()
            
        def forward(self, embeddings, file_ids, labels):
            # embeddings: (B, D)
            # file_ids: (B,) - which file each sample comes from
            # labels: (B,)
            
            # For each file, compute variance of embeddings from that file
            unique_files = torch.unique(file_ids)
            consistency_loss = 0.0
            count = 0
            
            for fid in unique_files:
                mask = (file_ids == fid)
                if mask.sum() > 1:  # Need at least 2 samples
                    file_embs = embeddings[mask]  # (n_samples, D)
                    file_labels = labels[mask]
                    
                    # Samples from same file should have consistent embeddings
                    # Measure variance
                    var = file_embs.var(dim=0).mean()
                    consistency_loss += var
                    count += 1
            
            if count > 0:
                consistency_loss = consistency_loss / count
            else:
                consistency_loss = torch.tensor(0.0, device=embeddings.device)
            
            return consistency_loss

    # -------------------------
    # MAIN MODEL: Frequency-Aware Adaptive Transformer (FAAT)
    # -------------------------
    class PatchEmbed1D(nn.Module):
        def __init__(self, patch_len, stride, d_model):
            super().__init__()
            self.patch_len = patch_len
            self.stride = stride
            self.proj = nn.Linear(patch_len, d_model)

        def forward(self, x):
            # x: (B, T)
            patches = x.unfold(dimension=1, size=self.patch_len, step=self.stride)  # (B, Np, patch_len)
            tok = self.proj(patches)  # (B, Np, d_model)
            
            # Compute local frequency indicators for FAPE
            freq_indicators = []
            for i in range(patches.shape[1]):
                patch = patches[:, i, :]  # (B, patch_len)
                # Simple high-frequency content measure: std of diff
                freq_ind = torch.std(torch.diff(patch, dim=-1), dim=-1)  # (B,)
                freq_indicators.append(freq_ind)
            freq_indicators = torch.stack(freq_indicators, dim=1)  # (B, Np)
            
            return tok, freq_indicators

    class TransformerBlock(nn.Module):
        def __init__(self, d_model, n_heads, dropout=0.1):
            super().__init__()
            self.norm1 = nn.LayerNorm(d_model)
            self.attn = nn.MultiheadAttention(d_model, n_heads, dropout=dropout, batch_first=True)
            self.drop1 = nn.Dropout(dropout)
            
            self.norm2 = nn.LayerNorm(d_model)
            self.mlp = nn.Sequential(
                nn.Linear(d_model, 4 * d_model),
                nn.GELU(),
                nn.Dropout(dropout),
                nn.Linear(4 * d_model, d_model),
                nn.Dropout(dropout),
            )

        def forward(self, x):
            x_norm = self.norm1(x)
            attn_out, _ = self.attn(x_norm, x_norm, x_norm)
            x = x + self.drop1(attn_out)
            x = x + self.mlp(self.norm2(x))
            return x

    class FrequencyAwareAdaptiveTransformer(nn.Module):
        def __init__(self, win_len, patch_len, patch_stride, d_model, n_heads, n_layers,
                     n_classes, n_freq_bands=8, dropout=0.1):
            super().__init__()
            
            # Patch embedding
            self.patch_embed = PatchEmbed1D(patch_len, patch_stride, d_model)
            
            # NOVEL: Adaptive Frequency Decomposition Module
            self.afdm = AdaptiveFrequencyDecomposition(win_len, n_bands=n_freq_bands)
            
            # NOVEL: Cross-Scale Spectral Fusion
            self.cssf = CrossScaleSpectralFusion(n_freq_bands, d_model)
            
            # NOVEL: Frequency-Adaptive Positional Encoding
            n_patches = 1 + (win_len - patch_len) // patch_stride
            self.fape = FrequencyAdaptivePositionalEncoding(d_model, max_len=n_patches)
            
            # Transformer blocks
            self.blocks = nn.ModuleList([
                TransformerBlock(d_model, n_heads, dropout=dropout)
                for _ in range(n_layers)
            ])
            
            self.norm = nn.LayerNorm(d_model)
            
            # Fusion of patch-level and spectral-level features
            self.fusion_proj = nn.Linear(2 * d_model, d_model)
            
            # Classification head
            self.head = nn.Linear(d_model, n_classes)
            
        def forward(self, x, return_features=False):
            # x: (B, T)
            B = x.shape[0]
            
            # 1. AFDM: Adaptive frequency decomposition
            afdm_out = self.afdm(x)
            band_energies = afdm_out['band_energies']  # (B, n_bands)
            
            # 2. CSSF: Cross-scale spectral fusion
            spectral_feat = self.cssf(band_energies)  # (B, d_model)
            
            # 3. Patch embedding
            tok, freq_ind = self.patch_embed(x)  # tok: (B, Np, d), freq_ind: (B, Np)
            
            # 4. FAPE: Frequency-adaptive positional encoding
            tok = self.fape(tok, freq_ind)
            
            # 5. Transformer blocks
            for blk in self.blocks:
                tok = blk(tok)
            tok = self.norm(tok)
            
            # 6. Global pooling
            patch_feat = tok.mean(dim=1)  # (B, d_model)
            
            # 7. Fuse patch-level and spectral-level features
            combined = torch.cat([patch_feat, spectral_feat], dim=-1)  # (B, 2*d_model)
            fused_feat = self.fusion_proj(combined)  # (B, d_model)
            
            # 8. Classification
            logits = self.head(fused_feat)
            
            if return_features:
                return logits, fused_feat, afdm_out
            return logits

    def compute_metrics(y_true, y_pred, n_classes):
        acc = accuracy_score(y_true, y_pred)
        prec, rec, f1, _ = precision_recall_fscore_support(y_true, y_pred, average='macro', zero_division=0)

    # -------------------------
    # MAIN TRAINING LOOP
    # -------------------------
    def train_model():
        # Build data
        train_ds, val_ds, actual_classes = build_data(
            ROOT, 
            CLASS_FOLDERS if not AUTO_DETECT_CLASSES else [], 
            VAL_SPLIT, 
            SIGNAL_KEY,
            FILE_EXTENSION
        )
        
        train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True, 
                                 num_workers=0, drop_last=True, pin_memory=True)
        val_loader = DataLoader(val_ds, batch_size=BATCH_SIZE, shuffle=False, 
                               num_workers=0, pin_memory=True)
        
        n_classes = len(actual_classes)
        
        # Initialize model
        model = FrequencyAwareAdaptiveTransformer(
            win_len=WIN_LEN,
            patch_len=PATCH_LEN,
            patch_stride=PATCH_STRIDE,
            d_model=D_MODEL,
            n_heads=N_HEADS,
            n_layers=N_LAYERS,
            n_classes=n_classes,
            n_freq_bands=N_FREQ_BANDS,
            dropout=DROPOUT,
        ).to(device)
        
        # Loss functions
        ce_loss = nn.CrossEntropyLoss()
        pi_spectral_loss = PhysicsInformedSpectralLoss().to(device)
        pgcl = PrototypeGuidedContrastive(n_classes, D_MODEL, temp=TEMP, momentum=PROTO_MOMENTUM).to(device)
        tscl = TemporalSpectralConsistency().to(device)
        
        # Optimizer with warmup
        optimizer = torch.optim.AdamW(model.parameters(), lr=LR, weight_decay=WEIGHT_DECAY)
        
        # Cosine schedule with warmup
        def lr_lambda(epoch):
            if epoch < WARMUP_EPOCHS:
                return epoch / WARMUP_EPOCHS
            else:
                return 0.5 * (1 + math.cos(math.pi * (epoch - WARMUP_EPOCHS) / (EPOCHS - WARMUP_EPOCHS)))
        
        scheduler = torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)
        
        # Training history
        history = {
            'train_loss': [], 'train_acc': [],
            'val_loss': [], 'val_acc': [],
            'val_f1': [],
        }
        
        best_val_f1 = -1.0
        best_path = "best_faat_model.pt"
        
        print("\n" + "="*80)
        print("TRAINING FREQUENCY-AWARE ADAPTIVE TRANSFORMER (FAAT)")
        print("="*80)
        
        for epoch in range(1, EPOCHS + 1):
            # ==================
            # TRAINING
            # ==================
            model.train()
            t0 = time.time()
            
            total_loss = 0.0
            total_ce = 0.0
            total_pi = 0.0
            total_pgcl = 0.0
            total_tscl = 0.0
            
            all_preds = []
            all_labels = []
            
            for x, y, file_ids in train_loader:
                x = x.to(device)
                y = y.to(device)
                file_ids = file_ids.to(device)
                
                optimizer.zero_grad(set_to_none=True)
                
                # Forward
                logits, features, afdm_out = model(x, return_features=True)
                
                # 1. Cross-entropy
                loss_ce = ce_loss(logits, y)
                
                # 2. Physics-informed spectral regularization
                loss_pi, pi_details = pi_spectral_loss(afdm_out, y, n_classes)
                
                # 3. Prototype-guided contrastive
                loss_pgcl = pgcl(features, y)
                
                # 4. Temporal-spectral consistency
                loss_tscl = tscl(features, file_ids, y)
                
                # Total loss
                loss = (loss_ce + 
                       LAMBDA_SPECTRAL_SPARSE * loss_pi + 
                       LAMBDA_SPECTRAL_CONTRAST * loss_pgcl +
                       LAMBDA_TEMPORAL_CONSIST * loss_tscl)
                
                loss.backward()
                nn.utils.clip_grad_norm_(model.parameters(), 1.0)
                optimizer.step()
                
                total_loss += loss.item()
                total_ce += loss_ce.item()
                total_pi += loss_pi.item()
                total_pgcl += loss_pgcl.item()
                total_tscl += loss_tscl.item()
                
                preds = logits.argmax(dim=-1)
                all_preds.append(preds.cpu().numpy())
                all_labels.append(y.cpu().numpy())
            
            scheduler.step()
            
            # Train metrics
            all_preds = np.concatenate(all_preds)
            all_labels = np.concatenate(all_labels)
            train_metrics = compute_metrics(all_labels, all_preds, n_classes)
            
            avg_loss = total_loss / len(train_loader)
            avg_ce = total_ce / len(train_loader)
            avg_pi = total_pi / len(train_loader)
            avg_pgcl = total_pgcl / len(train_loader)
            avg_tscl = total_tscl / len(train_loader)
            
            # ==================
            # VALIDATION
            # ==================
            model.eval()
            val_loss = 0.0
            val_preds = []
            val_labels = []
            
            with torch.no_grad():
                for x, y, _ in val_loader:
                    x = x.to(device)
                    y = y.to(device)
                    
                    logits = model(x)
                    loss = ce_loss(logits, y)
                    
                    val_loss += loss.item()
                    preds = logits.argmax(dim=-1)
                    val_preds.append(preds.cpu().numpy())
                    val_labels.append(y.cpu().numpy())
            
            val_preds = np.concatenate(val_preds)
            val_labels = np.concatenate(val_labels)
            val_metrics = compute_metrics(val_labels, val_preds, n_classes)
            
            avg_val_loss = val_loss / len(val_loader)
            
            # Save history
            history['train_loss'].append(avg_loss)
            history['train_acc'].append(train_metrics['accuracy'])
            history['val_loss'].append(avg_val_loss)
            history['val_acc'].append(val_metrics['accuracy'])
            history['val_f1'].append(val_metrics['f1_macro'])
            
            # Print progress
            elapsed = time.time() - t0
            print(f"Epoch {epoch:03d}/{EPOCHS} | "
                  f"LR: {get_lr(optimizer):.2e} | "
                  f"Time: {elapsed:.1f}s")
            print(f"  Train - Loss: {avg_loss:.4f} (CE: {avg_ce:.4f}, PI: {avg_pi:.4f}, "
                  f"PGCL: {avg_pgcl:.4f}, TSCL: {avg_tscl:.4f})")
            print(f"  Train - Acc: {train_metrics['accuracy']:.4f}, F1: {train_metrics['f1_macro']:.4f}")
            print(f"  Val   - Loss: {avg_val_loss:.4f}, Acc: {val_metrics['accuracy']:.4f}, "
                  f"F1: {val_metrics['f1_macro']:.4f}")
            
            # Save best model
            if val_metrics['f1_macro'] > best_val_f1:
                best_val_f1 = val_metrics['f1_macro']
                torch.save({
                    'epoch': epoch,
                    'model_state_dict': model.state_dict(),
                    'optimizer_state_dict': optimizer.state_dict(),
                    'val_f1': best_val_f1,
                    'config': {
                        'WIN_LEN': WIN_LEN,
                        'PATCH_LEN': PATCH_LEN,
                        'PATCH_STRIDE': PATCH_STRIDE,
                        'D_MODEL': D_MODEL,
                        'N_HEADS': N_HEADS,
                        'N_LAYERS': N_LAYERS,
                        'N_FREQ_BANDS': N_FREQ_BANDS,
                        'classes': actual_classes,
                    }
                }, best_path)
                print(f"  ✓ Best model saved (F1: {best_val_f1:.4f})")
            print()
        
        return model, history, val_loader, best_path

    def load_mat_signal(path):
        md = sio.loadmat(path)
        if 'signal' in md:
            arr = md['signal']
            if arr.ndim == 2 and arr.shape[0] < arr.shape[1]:
                if not hasattr(load_mat_signal, '_warned'):
                    print(f"  Multi-channel signal detected ({arr.shape[0]} channels x {arr.shape[1]} samples), using first channel")
                    load_mat_signal._warned = True
                arr = arr[0, :]
            return np.squeeze(arr).astype(np.float32)
        raise ValueError(f"No signal in {path}")

    def zscore(x, eps=1e-8):
        return (x - x.mean()) / (x.std() + eps)

    def window_signal(sig, win_len, stride, max_windows=None):
        n = sig.size
        if n < win_len:
            sig = np.pad(sig, (0, win_len - n), mode="constant")
            n = sig.size
        starts = np.arange(0, n - win_len + 1, stride)
        if max_windows is not None and starts.size > max_windows:
            idx = np.linspace(0, starts.size - 1, max_windows).astype(int)
            starts = starts[idx]
        return np.stack([sig[s:s+win_len] for s in starts], axis=0)

    # Augmentations
    def aug_time_shift(x, max_shift=200):
        return np.roll(x, np.random.randint(-max_shift, max_shift + 1))

    def aug_amp_scale(x, min_s=0.8, max_s=1.2):
        return x * np.random.uniform(min_s, max_s)

    def aug_add_noise(x, snr_db_min=20, snr_db_max=35):
        snr_db = np.random.uniform(snr_db_min, snr_db_max)
        sig_pow = np.mean(x**2) + 1e-12
        noise_pow = sig_pow / (10 ** (snr_db / 10.0))
        return x + np.random.randn(*x.shape).astype(np.float32) * np.sqrt(noise_pow)

    def apply_augs(x):
        if np.random.rand() < 0.5: x = aug_time_shift(x)
        if np.random.rand() < 0.5: x = aug_amp_scale(x)
        if np.random.rand() < 0.6: x = aug_add_noise(x)
        return x

    class AdaptiveFrequencyDecomposition(nn.Module):
        def __init__(self, signal_len, n_bands=8):
            super().__init__()
            self.n_bands = n_bands
            self.freq_bins = signal_len // 2 + 1
            self.band_centers = nn.Parameter(torch.linspace(0.05, 0.95, n_bands))
            self.band_log_widths = nn.Parameter(torch.full((n_bands,), -1.0))
            self.band_importance = nn.Parameter(torch.ones(n_bands))
        
        def forward(self, x):
            B, T = x.shape
            X = torch.fft.rfft(x, dim=-1)
            mag = torch.abs(X)
            F = mag.shape[-1]
            freqs = torch.linspace(0, 1, F, device=x.device)
            
            centers = torch.sigmoid(self.band_centers)
            widths = torch.exp(self.band_log_widths)
            band_masks = torch.exp(-((freqs.view(1, -1) - centers.view(-1, 1)) ** 2) / (2 * widths.view(-1, 1) ** 2 + 1e-8))
            band_masks = band_masks / (band_masks.sum(dim=1, keepdim=True) + 1e-8)
            
            importance = torch.softmax(self.band_importance, dim=0)
            weighted_bands = mag.unsqueeze(1) * band_masks.unsqueeze(0) * importance.view(1, -1, 1)
            band_energies = weighted_bands.sum(dim=-1)
            
            band_signals = []
            for b in range(self.n_bands):
                X_band = X * band_masks[b].unsqueeze(0)
                x_band = torch.fft.irfft(X_band, n=T, dim=-1)
                band_signals.append(x_band)
            band_signals = torch.stack(band_signals, dim=1)
            
            return {'band_energies': band_energies, 'band_signals': band_signals,
                    'full_spectrum': mag, 'band_masks': band_masks}

    class CrossScaleSpectralFusion(nn.Module):
        def __init__(self, n_bands, d_model):
            super().__init__()
            self.band_proj = nn.Linear(n_bands, d_model)
            self.fusion_attn = nn.MultiheadAttention(d_model, num_heads=4, batch_first=True)
            self.norm = nn.LayerNorm(d_model)
        
        def forward(self, band_energies):
            x = self.band_proj(band_energies).unsqueeze(1)
            x_fused, _ = self.fusion_attn(x, x, x)
            return self.norm(x + x_fused).squeeze(1)

    class FrequencyAdaptivePositionalEncoding(nn.Module):
        def __init__(self, d_model, max_len=100):
            super().__init__()
            pe = torch.zeros(max_len, d_model)
            position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
            div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
            pe[:, 0::2] = torch.sin(position * div_term)
            pe[:, 1::2] = torch.cos(position * div_term)
            self.register_buffer('pe', pe)
            self.freq_modulation = nn.Linear(1, d_model)
        
        def forward(self, x, freq_indicator):
            B, N, d = x.shape
            pos_enc = self.pe[:N, :].unsqueeze(0).expand(B, -1, -1)
            freq_mod = self.freq_modulation(freq_indicator.unsqueeze(-1))
            return x + pos_enc * (1 + torch.tanh(freq_mod))

    class PhysicsInformedSpectralLoss(nn.Module):
        def forward(self, afdm_out, labels, n_classes):
            band_energies = afdm_out['band_energies']
            band_masks = afdm_out['band_masks']
            
            sparsity = torch.mean(torch.abs(band_energies))
            
            band_mask_norm = band_masks / (band_masks.sum(dim=1, keepdim=True) + 1e-8)
            localization = -torch.mean(torch.sum(band_mask_norm * torch.log(band_mask_norm + 1e-8), dim=1))
            localization = -localization
            
            class_means = []
            for c in range(n_classes):
                mask = (labels == c)
                if mask.any():
                    class_means.append(band_energies[mask].mean(dim=0))
            
            if len(class_means) > 1:
                class_means = torch.stack(class_means)
                dists = torch.cdist(class_means, class_means, p=2)
                separation = -torch.mean(dists[dists > 0])
            else:
                separation = torch.tensor(0.0, device=band_energies.device)
            
            return sparsity + 0.1 * localization + 0.2 * separation, {}

    class PrototypeGuidedContrastive(nn.Module):
        def __init__(self, n_classes, embed_dim, temp=0.07, momentum=0.95):
            super().__init__()
            self.n_classes = n_classes
            self.temp = temp
            self.momentum = momentum
            self.register_buffer('prototypes', torch.randn(n_classes, embed_dim))
            self.prototypes = F.normalize(self.prototypes, dim=-1)
            self.register_buffer('initialized', torch.zeros(n_classes, dtype=torch.bool))
        
        @torch.no_grad()
        def update_prototypes(self, embeddings, labels):
            for c in range(self.n_classes):
                mask = (labels == c)
                if mask.any():
                    mean_c = F.normalize(embeddings[mask].mean(dim=0), dim=-1)
                    if not self.initialized[c]:
                        self.prototypes[c] = mean_c
                        self.initialized[c] = True
                    else:
                        self.prototypes[c] = self.momentum * self.prototypes[c] + (1 - self.momentum) * mean_c
                        self.prototypes[c] = F.normalize(self.prototypes[c], dim=-1)
        
        def forward(self, embeddings, labels):
            embeddings = F.normalize(embeddings, dim=-1)
            with torch.no_grad():
                self.update_prototypes(embeddings, labels)
            prototypes = F.normalize(self.prototypes, dim=-1)
            logits = (embeddings @ prototypes.t()) / self.temp
            return F.cross_entropy(logits, labels)

    class TemporalSpectralConsistency(nn.Module):
        def forward(self, embeddings, file_ids, labels):
            unique_files = torch.unique(file_ids)
            consistency_loss = 0.0
            count = 0
            for fid in unique_files:
                mask = (file_ids == fid)
                if mask.sum() > 1:
                    consistency_loss += embeddings[mask].var(dim=0).mean()
                    count += 1
            return consistency_loss / count if count > 0 else torch.tensor(0.0, device=embeddings.device)

    class PatchEmbed1D(nn.Module):
        def __init__(self, patch_len, stride, d_model):
            super().__init__()
            self.patch_len = patch_len
            self.stride = stride
            self.proj = nn.Linear(patch_len, d_model)
        
        def forward(self, x):
            patches = x.unfold(dimension=1, size=self.patch_len, step=self.stride)
            tok = self.proj(patches)
            freq_indicators = []
            for i in range(patches.shape[1]):
                freq_ind = torch.std(torch.diff(patches[:, i, :], dim=-1), dim=-1)
                freq_indicators.append(freq_ind)
            freq_indicators = torch.stack(freq_indicators, dim=1)
            return tok, freq_indicators

    class TransformerBlock(nn.Module):
        def __init__(self, d_model, n_heads, dropout=0.1):
            super().__init__()
            self.norm1 = nn.LayerNorm(d_model)
            self.attn = nn.MultiheadAttention(d_model, n_heads, dropout=dropout, batch_first=True)
            self.drop1 = nn.Dropout(dropout)
            self.norm2 = nn.LayerNorm(d_model)
            self.mlp = nn.Sequential(
                nn.Linear(d_model, 4 * d_model), nn.GELU(), nn.Dropout(dropout),
                nn.Linear(4 * d_model, d_model), nn.Dropout(dropout))
        
        def forward(self, x):
            x_norm = self.norm1(x)
            attn_out, _ = self.attn(x_norm, x_norm, x_norm)
            x = x + self.drop1(attn_out)
            x = x + self.mlp(self.norm2(x))
            return x

    class FrequencyAwareAdaptiveTransformer(nn.Module):
        def __init__(self, win_len, patch_len, patch_stride, d_model, n_heads, n_layers,
                     n_classes, n_freq_bands=8, dropout=0.1):
            super().__init__()
            self.patch_embed = PatchEmbed1D(patch_len, patch_stride, d_model)
            self.afdm = AdaptiveFrequencyDecomposition(win_len, n_bands=n_freq_bands)
            self.cssf = CrossScaleSpectralFusion(n_freq_bands, d_model)
            
            n_patches = 1 + (win_len - patch_len) // patch_stride
            self.fape = FrequencyAdaptivePositionalEncoding(d_model, max_len=n_patches)
            
            self.blocks = nn.ModuleList([
                TransformerBlock(d_model, n_heads, dropout=dropout) for _ in range(n_layers)])
            self.norm = nn.LayerNorm(d_model)
            self.fusion_proj = nn.Linear(2 * d_model, d_model)
            self.head = nn.Linear(d_model, n_classes)
        
        def forward(self, x, return_features=False):
            afdm_out = self.afdm(x)
            spectral_feat = self.cssf(afdm_out['band_energies'])
            
            tok, freq_ind = self.patch_embed(x)
            tok = self.fape(tok, freq_ind)
            
            for blk in self.blocks:
                tok = blk(tok)
            tok = self.norm(tok)
            
            patch_feat = tok.mean(dim=1)
            combined = torch.cat([patch_feat, spectral_feat], dim=-1)
            fused_feat = self.fusion_proj(combined)
            logits = self.head(fused_feat)
            
            if return_features:
                return logits, fused_feat, afdm_out
            return logits

    # -------------------------
    # TRAINING
    # -------------------------
    def compute_metrics(y_true, y_pred, n_classes):
        acc = accuracy_score(y_true, y_pred)
        prec, rec, f1, _ = precision_recall_fscore_support(y_true, y_pred, average='macro', zero_division=0)
        return {'accuracy': acc, 'precision_macro': prec, 'recall_macro': rec, 'f1_macro': f1}

    paths, ys = build_index(ROOT, CLASS_FOLDERS, MAX_FILES_PER_CLASS)
    print(f"\nTotal files: {len(paths)}")

    train_paths, val_paths, train_y, val_y = train_test_split(paths, ys, test_size=VAL_SPLIT, random_state=SEED, stratify=ys)

    train_ds = PumpDataset(train_paths, train_y, WIN_LEN, WIN_STRIDE, MAX_WINDOWS_PER_FILE, train=True)
    val_ds = PumpDataset(val_paths, val_y, WIN_LEN, WIN_STRIDE, MAX_WINDOWS_PER_FILE, train=False)

    print(f"Train: {len(train_ds)} windows, Val: {len(val_ds)} windows\n")

    train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True, num_workers=0, drop_last=True)
    val_loader = DataLoader(val_ds, batch_size=BATCH_SIZE, shuffle=False, num_workers=0)

    n_classes = len(CLASS_FOLDERS)
    model = FrequencyAwareAdaptiveTransformer(WIN_LEN, PATCH_LEN, PATCH_STRIDE, D_MODEL, N_HEADS, N_LAYERS,
                                             n_classes, N_FREQ_BANDS, DROPOUT).to(device)

    ce_loss = nn.CrossEntropyLoss()
    pi_loss = PhysicsInformedSpectralLoss().to(device)
    pgcl = PrototypeGuidedContrastive(n_classes, D_MODEL, TEMP, PROTO_MOMENTUM).to(device)
    tscl = TemporalSpectralConsistency().to(device)

    optimizer = torch.optim.AdamW(model.parameters(), lr=LR, weight_decay=WEIGHT_DECAY)
    scheduler = torch.optim.lr_scheduler.LambdaLR(optimizer, 
        lambda e: e/WARMUP_EPOCHS if e < WARMUP_EPOCHS else 0.5*(1+math.cos(math.pi*(e-WARMUP_EPOCHS)/(EPOCHS-WARMUP_EPOCHS))))

    history = {'train_loss': [], 'train_acc': [], 'val_loss': [], 'val_acc': [], 'val_f1': []}
    best_val_f1 = -1.0
    best_path = os.path.join(OUTPUT_DIR, "best_faat_3bar_model.pt")

    print("="*80)
    print("TRAINING STARTED")
    print("="*80)

    for epoch in range(1, EPOCHS + 1):
        model.train()
        t0 = time.time()
        total_loss = 0.0
        all_preds, all_labels = [], []
        
        for x, y, file_ids in train_loader:
            x, y, file_ids = x.to(device), y.to(device), file_ids.to(device)
            optimizer.zero_grad(set_to_none=True)
            
            logits, features, afdm_out = model(x, return_features=True)
            loss_ce = ce_loss(logits, y)
            loss_pi, _ = pi_loss(afdm_out, y, n_classes)
            loss_pgcl = pgcl(features, y)
            loss_tscl = tscl(features, file_ids, y)
            
            loss = loss_ce + LAMBDA_SPECTRAL_SPARSE * loss_pi + LAMBDA_SPECTRAL_CONTRAST * loss_pgcl + LAMBDA_TEMPORAL_CONSIST * loss_tscl
            loss.backward()
            nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            optimizer.step()
            
            total_loss += loss.item()
            all_preds.append(logits.argmax(dim=-1).cpu().numpy())
            all_labels.append(y.cpu().numpy())
        
        scheduler.step()
        
        all_preds = np.concatenate(all_preds)
        all_labels = np.concatenate(all_labels)
        train_metrics = compute_metrics(all_labels, all_preds, n_classes)
        
        # Validation
        model.eval()
        val_loss = 0.0
        val_preds, val_labels = [], []
        with torch.no_grad():
            for x, y, _ in val_loader:
                x, y = x.to(device), y.to(device)
                logits = model(x)
                val_loss += ce_loss(logits, y).item()
                val_preds.append(logits.argmax(dim=-1).cpu().numpy())
                val_labels.append(y.cpu().numpy())
        
        val_preds = np.concatenate(val_preds)
        val_labels = np.concatenate(val_labels)
        val_metrics = compute_metrics(val_labels, val_preds, n_classes)
        
        history['train_loss'].append(total_loss / len(train_loader))
        history['train_acc'].append(train_metrics['accuracy'])
        history['val_loss'].append(val_loss / len(val_loader))
        history['val_acc'].append(val_metrics['accuracy'])
        history['val_f1'].append(val_metrics['f1_macro'])
        
        elapsed = time.time() - t0
        print(f"Epoch {epoch:02d}/{EPOCHS} | Time: {elapsed:.1f}s | "
              f"Train Acc: {train_metrics['accuracy']:.4f} | Val Acc: {val_metrics['accuracy']:.4f} | "
              f"Val F1: {val_metrics['f1_macro']:.4f}")
        
        if val_metrics['f1_macro'] > best_val_f1:
            best_val_f1 = val_metrics['f1_macro']
            torch.save({'model': model.state_dict(), 'classes': CLASS_FOLDERS}, best_path)
            print(f"  ✓ Best model saved (F1: {best_val_f1:.4f})")

    # Get predictions, probabilities, features
    all_preds, all_labels, all_probs, all_features = [], [], [], []
    with torch.no_grad():
        for x, y, _ in val_loader:
            x = x.to(device)
            logits, features, _ = model(x, return_features=True)
            probs = F.softmax(logits, dim=-1)
            all_preds.append(logits.argmax(dim=-1).cpu().numpy())

            all_probs.append(probs.cpu().numpy())
            all_features.append(features.cpu().numpy())

    y_true = np.concatenate(all_labels)
